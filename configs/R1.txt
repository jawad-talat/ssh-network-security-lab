! ============================================================
! Device: R1 — HQ Router
! Role: Router-on-a-Stick, SSH, Static Routing, ACL
! ============================================================

hostname R1

ip domain-name benzi.local
ip ssh version 2
username admin secret [password]
crypto key generate rsa general-keys modulus 2048

! Subinterfaces
interface GigabitEthernet0/0
 no ip address
 no shutdown

interface GigabitEthernet0/0.10
 encapsulation dot1Q 10
 ip address 192.168.10.1 255.255.255.0
 ip access-group BLOCK_SSH in

interface GigabitEthernet0/0.20
 encapsulation dot1Q 20
 ip address 192.168.20.1 255.255.255.0

! Serial WAN — DCE side sets clock rate
interface Serial0/3/0
 ip address 10.0.0.1 255.255.255.252
 clock rate 64000

! Static Routes
ip route 10.0.0.2 255.255.255.255 Serial0/3/0
ip route 192.168.30.0 255.255.255.0 10.0.0.2

! ACL — permit Admin-PC SSH only, block User-PC
ip access-list extended BLOCK_SSH
 permit tcp host 192.168.20.10 any eq 22
 deny tcp 192.168.30.0 0.0.0.255 any eq 22
 permit ip any any

line vty 0 4
 login local
 transport input ssh
