! ============================================================
! Device: R2 — Branch Router
! Role: SSH, Static Routing, ACL
! ============================================================

hostname R2

ip domain-name benzi.local
ip ssh version 2
username admin secret [password]
crypto key generate rsa general-keys modulus 2048

! LAN interface facing User-PC
interface GigabitEthernet0/0
 ip address 192.168.30.1 255.255.255.0
 ip access-group BLOCK_SSH in
 no shutdown

! Serial WAN — DTE side, no clock rate
interface Serial0/3/0
 ip address 10.0.0.2 255.255.255.252

! Static Routes
ip route 192.168.10.0 255.255.255.0 10.0.0.1
ip route 192.168.20.0 255.255.255.0 10.0.0.1

! ACL — same policy as R1
ip access-list extended BLOCK_SSH
 permit tcp host 192.168.20.10 any eq 22
 deny tcp 192.168.30.0 0.0.0.255 any eq 22
 permit ip any any

line vty 0 4
 login local
 transport input ssh
